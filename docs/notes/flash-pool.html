为什么不是只改一个 hash

底层 MemCache 可以保存字节，但它不知道这些字节属于哪个层、哪个 head、哪个状态边界。跨实例共享要同时回答：找哪个对象？对象是什么格式？读回本地哪里？什么时候才可读？只让两个 key 相等，不能保证 payload 兼容。

    def objects(self, start: int, block_id: int, digest: str) -> tuple[PoolObject, ...]:
        if not 0 <= block_id < self.num_blocks or start < 0 or start % self.chunk_size:
            raise ValueError("Invalid local block or Store chunk boundary")
        offset = start % self.block_size
        chunks = [(offset, offset + self.chunk_size)] * len(self.local_shards)
        addresses, sizes, _ = self.layout.prepare_values(chunks, [block_id], self.local_shards)
        owners = [
            (name, layout.prepare_values(chunks, [block_id], self.local_shards))
            for name, layout in self.layer_layouts.items()
        ]
        result = []
        for index, shard in enumerate(self.local_shards):
            ranges, actual = [], []
            cursor = 0
            for name, (layer_addrs, layer_sizes, _) in owners:
                for address, size in zip(layer_addrs[index], layer_sizes[index], strict=True):
                    ranges.append(PoolRange(name, int(address), cursor, int(size)))
                    actual.append((address, size))
                    cursor += int(size)
            if actual != list(zip(addresses[index], sizes[index], strict=True)):
                raise ValueError("Layer ranges disagree with the group's object byte order")
            result.append(PoolObject(self.key(shard, digest), cursor, block_id, tuple(ranges)))
        return tuple(result)

    def lookup(self, token_len, hashes):
        if not self.layouts:
            return 0
        common = None
        for layout in self.layouts.values():
            hits = set()
            for start in range(0, token_len - layout.chunk_size + 1, layout.chunk_size):
                end = start + layout.chunk_size
                hash_index = end // layout.hash_block_size - 1
                if hash_index >= len(hashes):
                    break
                digest = block_hash_to_str(hashes[hash_index])
                keys = [layout.key(shard, digest) for shard in range(layout.store_tp)]
                if not all(self.io.batch_is_exist(keys)):
                    if layout.state:
                        # State checkpoints can be sparse; unlike attention,
                        # they do not require every historical block to exist.
                        continue
                    break
                hits.add(end)
            common = hits if common is None else common.intersection(hits)
        alignment = self.worker.cache_transfer_granularity
        # Rounding a checkpoint down would claim a state we never restored.
        return max((end for end in common or () if end % alignment == 0), default=0)

use_layerwise=true 控制 P 写入的逐层时机；PD 的 D 仍整体读取完成后恢复计算。普通前缀池读取可以按层或整体执行。这三件事不能用一个“异步”词概括。

发布与租约：分配成功还不算命中

写：batch_is_readable → batch_alloc → 逐层/整体 batch_copy
    → drain → 校验所有 range 完成 → batch_write_finish → 释放租约

读：batch_add_lease → batch_get_key_info（地址与长度）
    → batch_copy → 完成读取 → 释放租约

查询使用 readable 状态，不以“已分配”代替“已写完”。batch_add_lease 返回状态码，不是对象地址。多个本地读取共享同一个 key 的租约引用；最后一个使用者结束后才释放。失败的未完成对象不能发布，也不贸然用删除操作补偿，以免误删其他 writer 共享的对象。

    def finish(self, session: StoreIOSession) -> None:
        if session.closed:
            return
        try:
            expected = sum(len(plan.ranges) for plan in session.plans)
            if len(session.completed) != expected:
                raise RuntimeError("Cannot finish a partially copied Store object")
            if time.monotonic() >= session.deadline:
                raise RuntimeError("Store lease expired before publication")
            keys = [plan.key for plan in session.plans]
            if session.writing and keys:
                results = self.backend.batch_write_finish(keys, [0] * len(keys))
                if len(results) != len(keys) or any(result != 0 for result in results):
                    raise RuntimeError("MemCache publication failed")
        except Exception as error:
            try:
                self.abort(session)
            except Exception as cleanup:
                error.add_note(f"Lease cleanup also failed: {cleanup}")
            raise
        self.abort(session)
